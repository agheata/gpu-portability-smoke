cmake_minimum_required(VERSION 3.21)
project(gpu_portability_smoke LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE) # some device compilers do not optimize by default
endif()

set(GPU_BACKEND "CUDA" CACHE STRING "Device backend: CUDA, HIP or CUSTOM (CUDA-like toolchain)")
set_property(CACHE GPU_BACKEND PROPERTY STRINGS CUDA HIP CUSTOM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# All device code is relocatable and linked into each executable, across static libraries (as in VecGeom/AdePT).
if(GPU_BACKEND STREQUAL "CUDA")
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(CMAKE_CUDA_SEPARABLE_COMPILATION ON)
  set(GPU_LANG CUDA)
  add_compile_definitions(GPU_BACKEND_CUDA)
  set(GPU_LINK_OPTIONS "")
  set(GPU_RUNTIME_LIBS "")
elseif(GPU_BACKEND STREQUAL "HIP")
  enable_language(HIP)
  find_package(hip REQUIRED)
  set(CMAKE_HIP_STANDARD 17)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  set(GPU_LANG HIP)
  add_compile_definitions(GPU_BACKEND_HIP)
  add_compile_options($<$<COMPILE_LANGUAGE:HIP>:-fgpu-rdc>)
  set(GPU_LINK_OPTIONS -fgpu-rdc --hip-link)
  foreach(_arch ${CMAKE_HIP_ARCHITECTURES})
    list(APPEND GPU_LINK_OPTIONS --offload-arch=${_arch})
  endforeach()
  set(GPU_RUNTIME_LIBS hip::host)
elseif(GPU_BACKEND STREQUAL "CUSTOM")
  set(GPU_RUNTIME_HEADER "" CACHE STRING "Runtime header of the CUDA-like toolchain, e.g. xyz_runtime.h")
  set(GPU_API_PREFIX "" CACHE STRING "Prefix of the runtime API, e.g. xyz for xyzMalloc")
  set(GPU_COMPILE_FLAGS "" CACHE STRING "Flags to compile the .cu sources as relocatable device code")
  set(GPU_LINK_FLAGS "" CACHE STRING "Flags to link executables, including the device link step")
  set(GPU_LIBRARIES "" CACHE STRING "Runtime libraries to link")
  if(NOT GPU_RUNTIME_HEADER OR NOT GPU_API_PREFIX)
    message(FATAL_ERROR "GPU_BACKEND=CUSTOM needs GPU_RUNTIME_HEADER and GPU_API_PREFIX")
  endif()
  set(GPU_LANG CXX)
  add_compile_definitions(GPU_BACKEND_CUSTOM "GPU_RUNTIME_HEADER=<${GPU_RUNTIME_HEADER}>"
                          GPU_API_PREFIX=${GPU_API_PREFIX})
  separate_arguments(_compile_flags UNIX_COMMAND "${GPU_COMPILE_FLAGS}")
  add_compile_options(${_compile_flags})
  separate_arguments(GPU_LINK_OPTIONS UNIX_COMMAND "${GPU_LINK_FLAGS}")
  set(GPU_RUNTIME_LIBS ${GPU_LIBRARIES})
else()
  message(FATAL_ERROR "GPU_BACKEND must be CUDA, HIP or CUSTOM")
endif()
message(STATUS "GPU backend: ${GPU_BACKEND}")
option(GPU_PURE_VIRTUAL_SHIM "Define a device __cxa_pure_virtual, for toolchains that do not provide one" OFF)
if(GPU_PURE_VIRTUAL_SHIM)
  add_compile_definitions(GPU_PURE_VIRTUAL_SHIM)
endif()

# gpu_target(<name> EXECUTABLE|STATIC SOURCES ... [LIBS ...] [OUTPUT_NAME name])
function(gpu_target name kind)
  cmake_parse_arguments(ARG "" "OUTPUT_NAME" "SOURCES;LIBS" ${ARGN})
  set_source_files_properties(${ARG_SOURCES} PROPERTIES LANGUAGE ${GPU_LANG})
  if(kind STREQUAL "STATIC")
    add_library(${name} STATIC ${ARG_SOURCES})
  else()
    add_executable(${name} ${ARG_SOURCES})
    target_link_options(${name} PRIVATE ${GPU_LINK_OPTIONS})
    target_link_libraries(${name} PRIVATE ${GPU_RUNTIME_LIBS})
    if(ARG_OUTPUT_NAME)
      set_target_properties(${name} PROPERTIES OUTPUT_NAME ${ARG_OUTPUT_NAME})
    endif()
  endif()
  target_include_directories(${name} PUBLIC ${PROJECT_SOURCE_DIR}/common)
  if(ARG_LIBS)
    target_link_libraries(${name} PUBLIC ${ARG_LIBS})
  endif()
endfunction()
